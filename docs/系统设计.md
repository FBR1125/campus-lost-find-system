# 

# \### 1. 系统架构

# 本系统采用 B/S（浏览器 / 服务器）架构，整体分为三层架构设计：

# 表现层（用户交互层）：基于 Vue3+Element Plus 实现，负责用户界面展示、交互操作接收与反馈，适配多终端设备的页面渲染；

# 业务逻辑层（核心处理层）：基于 Java+Spring Boot 实现，负责接收表现层请求，处理失物登记、招领发布、审核、认领等核心业务逻辑，实现权限校验、数据校验等功能；

# 数据访问层（数据库交互层）：基于 MyBatis-Plus 实现，负责业务逻辑层与数据库的交互，完成数据的增删改查操作，保证数据一致性。

# 

# \### 2. 模块设计

# （说明主要模块及其职责关系）

# 模块名称	模块职责

# 用户模块	负责用户注册、登录、密码找回、个人中心管理、权限校验等

# 失物管理模块	负责普通用户失物信息登记、提交，管理员失物信息审核、展示管理等

# 招领管理模块	负责普通用户招领信息发布、提交，管理员招领信息审核、展示管理等

# 认领管理模块	负责普通用户认领申请提交，管理员认领申请审核、认领结果通知等

# 系统管理模块	仅管理员可操作，负责用户账号管理、系统日志查看、基础配置调整等

# 公共查询模块	负责失物 / 招领信息的检索、分类展示、首页信息推荐等

# 

# \### 3. 数据库设计

# \- E-R 图（）

# 核心实体及关系：

# 用户（用户 ID、账号、密码、姓名、联系方式、角色类型）：作为核心实体，普通用户可关联多条失物信息、招领信息、认领记录；管理员可关联多条审核记录；

# 失物信息（失物 ID、用户 ID、物品名称、类型、遗失地点 / 时间、特征、审核状态）：与用户为多对一关系；

# 招领信息（招领 ID、用户 ID、物品名称、类型、拾获地点 / 时间、特征、审核状态）：与用户为多对一关系；

# 认领记录（认领 ID、招领 ID、用户 ID、认领依据、审核状态、审核结果）：与招领信息为多对一关系，与用户为多对一关系。

字段说明——用户表:user

字段名	类型	为空	默认值	字段说明
id	int	NO	自增	主键 ID
username	varchar(50)	NO	无	用户名（唯一）
password	varchar(100)	NO	无	密码
role	varchar(20)	YES	user	角色：user = 普通用户，admin = 管理员
avatar	varchar(255)	YES	NULL	头像路径
create_time	datetime	YES	CURRENT_TIMESTAMP	创建时间
物品信息表:item

字段名	类型	为空	默认值	字段说明
id	int	NO	自增	物品 ID
user_id	int	NO	无	发布人 ID（关联 user 表）
name	varchar(100)	NO	无	物品名称
place	varchar(255)	YES	NULL	丢失 / 捡到地点
description	text	YES	NULL	物品详细描述
contact	varchar(50)	YES	NULL	联系方式
type	int	NO	无	物品类型：0 = 丢失，1 = 捡到
status	int	YES	0	认领状态：0 = 待认领，1 = 已认领（仅招领使用）
found_status	int	YES	0	找回状态：0 = 待找回，1 = 已找回（仅失物使用）
check_status	int	YES	0	审核状态：0 = 待审核，1 = 审核通过，2 = 不通过
create_time	datetime	YES	CURRENT_TIMESTAMP	发布时间
认领记录表:claim

字段名	类型	为空	默认值	字段说明
id	int	NO	自增	记录 ID
item_id	int	NO	无	物品 ID
user_id	int	NO	无	认领用户 ID
message	varchar(255)	YES	NULL	认领备注 / 说明
create_time	datetime	YES	CURRENT_TIMESTAMP	认领时间
建表sql——CREATE TABLE user ( id int NOT NULL AUTO_INCREMENT, username varchar(50) NOT NULL COMMENT '用户名', password varchar(100) NOT NULL COMMENT '密码', role varchar(20) DEFAULT 'user' COMMENT '角色：user/admin', avatar varchar(255) DEFAULT NULL COMMENT '头像地址', create_time datetime DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY username (username) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

CREATE TABLE item ( id int NOT NULL AUTO_INCREMENT, user_id int NOT NULL COMMENT '发布人ID', name varchar(100) NOT NULL COMMENT '物品名称', place varchar(255) DEFAULT NULL COMMENT '丢失/捡到地点', description text COMMENT '物品描述', contact varchar(50) DEFAULT NULL COMMENT '联系方式', type int NOT NULL COMMENT '0=丢失 1=捡到', status int DEFAULT '0' COMMENT '0=待认领 1=已认领', found_status int DEFAULT '0' COMMENT '0=待找回 1=已找回', check_status int DEFAULT '0' COMMENT '0=待审核 1=审核通过 2=不通过', create_time datetime DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY user_id (user_id) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='失物招领物品表';

CREATE TABLE claim ( id int NOT NULL AUTO_INCREMENT, item_id int NOT NULL COMMENT '物品ID', user_id int NOT NULL COMMENT '认领人ID', message varchar(255) DEFAULT NULL COMMENT '认领说明', create_time datetime DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY item_id (item_id), KEY user_id (user_id) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='物品认领记录表';
